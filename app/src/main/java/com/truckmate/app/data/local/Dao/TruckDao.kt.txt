package com.truckmate.app.data.local.Dao

import androidx.room.Dao
import androidx.room.Insert
import androidx.room.OnConflictStrategy
import androidx.room.Query
import com.truckmate.app.data.local.entities.TruckEntity
import kotlinx.coroutines.flow.Flow

@Dao
interface TruckDao {
    @Query("SELECT * FROM trucks")
    fun getAllTrucks(): Flow<List<TruckEntity>>

    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertAll(trucks: List<TruckEntity>)

    @Query("SELECT * FROM trucks WHERE id = :id LIMIT 1")
    suspend fun getTruckById(id: String): TruckEntity?
}
