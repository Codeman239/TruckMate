package com.truckmate.app.ui.screens

import androidx.compose.foundation.clickable
import androidx.compose.foundation.layout.*
import androidx.compose.foundation.lazy.LazyColumn
import androidx.compose.foundation.lazy.items
import androidx.compose.material3.*
import androidx.compose.runtime.*
import androidx.compose.ui.Modifier
import androidx.compose.ui.unit.dp
import com.truckmate.app.data.local.entities.TruckEntity
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.asStateFlow

/**
 * HomeScreen shows a map + a list of nearby trucks. This simplified version
 * shows a list of trucks and a search bar. The real map integration (Google Maps)
 * is left as a clear placeholder and instructions are provided below.
 */
@Composable
fun HomeScreen(onOpenDetails: (String) -> Unit) {
    // Simple sample data — replace by ViewModel/Room flow in real app
    val sample = listOf(
        TruckEntity("truck1", "Tasty Tacos", 40.0, -74.0, "Best tacos in town", true, 4.8),
        TruckEntity("truck2", "Burger Bus", 40.01, -74.01, "Gourmet burgers", true, 4.5)
    )

    var query by remember { mutableStateOf("") }

    Column(Modifier.fillMaxSize().padding(12.dp)) {
        OutlinedTextField(value = query, onValueChange = { query = it }, label = { Text("Search trucks or food") })
        Spacer(modifier = Modifier.height(8.dp))
        Text("Nearby trucks", style = MaterialTheme.typography.titleMedium)
        Spacer(modifier = Modifier.height(8.dp))
        LazyColumn {
            items(sample.filter { it.name.contains(query, true) || it.description?.contains(query, true) == true }) { truck ->
                TruckListItem(truck = truck, onClick = { onOpenDetails(truck.id) })
            }
        }
    }
}

@Composable
fun TruckListItem(truck: TruckEntity, onClick: () -> Unit) {
    Card(modifier = Modifier.fillMaxWidth().padding(vertical = 4.dp).clickable { onClick() }) {
        Row(modifier = Modifier.padding(12.dp), horizontalArrangement = Arrangement.SpaceBetween) {
            Column {
                Text(truck.name, style = MaterialTheme.typography.titleMedium)
                Text(truck.description ?: "", style = MaterialTheme.typography.bodySmall)
            }
            Column(horizontalAlignment = androidx.compose.ui.Alignment.End) {
                Text(if (truck.isOpen) "Open" else "Closed")
                Text("⭐ ${truck.rating}")
            }
        }
    }
}
